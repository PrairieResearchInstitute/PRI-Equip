# PaCELab Equip: every setting the application reads from its environment.
# Pattern: drez-config-secrets templates\env.template.txt (platform D-003, PHP track).
#
# Where the application looks, first source that sets a name winning
# (includes/env.php, D-010):
#   1. the real environment (a server's service file, compose env_file);
#   2. the file PACELAB_ENV_FILE names, when set;
#   3. pg.env beside the data folder: on the drive <estate>\data\pacelab-equip\pg.env,
#      found from config.local.json dataFolder (platform\scripts\pg-provision.ps1 writes it);
#   4. .env beside index.php.
# Readable only by the account that runs the application; never in git.
#
# The driver: PACELAB_DRIVER in the real environment wins; otherwise databaseDriver in
# config.local.json (php tools/config.php set databaseDriver pgsql); otherwise a
# PACELAB_DRIVER line in one of the files; otherwise SQLite in the data folder.

# sqlite (the default) or pgsql
PACELAB_DRIVER=
# PostgreSQL connection, used only when the driver is pgsql. On the drive the
# estate cluster answers on PGHOST=127.0.0.1, PGPORT=5433 (platform D-028), and
# those are the defaults when the names are empty.
PGHOST=
PGPORT=
PGDATABASE=
PGUSER=
# Secret: the application role's password. Never commit a filled copy.
PGPASSWORD=
# The throwaway test database; its name must end in _test. The tests drop every
# table in it and rebuild it from migrations/*.pgsql.sql on each run.
PGTESTDATABASE=
# TLS (Radiant audit M3): sslmode verify-full. compose.yaml mounts the NCSA CA
# file pg-ca.crt (copied beside compose.yaml) read-only at /run/secrets/pg-ca.crt.
# libpq, under pdo_pgsql, reads PGSSLROOTCERT for the CA file and, with sslmode
# verify-full, checks the certificate chain and the host name. Keep the value as
# written; libpq ignores a missing file unless the sslmode is verify-ca or verify-full.
PGSSLROOTCERT=/run/secrets/pg-ca.crt
# The owner migration runs through docker run, outside compose.yaml, so it needs
# the mount too: add -v "$PWD/pg-ca.crt:/run/secrets/pg-ca.crt:ro" to that command.

# Tests only (tests/run-tests.php): pgsql runs the suite against the test database.
# PACELAB_TEST_PGDATABASE, when set, names the test database ahead of PGTESTDATABASE.
PACELAB_TEST_DRIVER=
PACELAB_TEST_PGDATABASE=
# Optional, for a copy outside the estate (a hosted server): the folder holding
# pg_dump and pg_restore. On the drive tools/pg-backup.php uses the estate's
# runtime, <estate>\runtime\postgresql\17\bin, ahead of this (platform D-028).
PACELAB_PG_BIN=
# Optional: one more environment file, read ahead of pg.env and .env.
PACELAB_ENV_FILE=

# Identity (D-015, includes/auth.php). None of these is a secret.
# password (the default): the application's own accounts and typed names.
# local: the development sign-in on the owner's computer (loopback, PHP's built-in
#        server, no proxy header); this computer sets it with
#        php tools/config.php set authMode local
# proxy: Radiant. The university's authenticating proxy says who (NetID); role_grants
#        (migration 0003, tools/roles.php) says what. The container image sets proxy.
PACELAB_AUTH_MODE=
# The header the proxy puts the verified identity in. An address keeps its local part
# as the NetID when its domain is listed in PACELAB_IDENTITY_DOMAINS; a bare value is
# taken as the NetID. Defaults: X-Forwarded-Email and illinois.edu. NCSA confirms both.
PACELAB_IDENTITY_HEADER=
PACELAB_IDENTITY_DOMAINS=
# The header carrying campus groups, comma separated (default X-Forwarded-Groups);
# none turns groups off.
PACELAB_GROUPS_HEADER=
# Optional: the addresses or CIDR ranges the proxy connects from; any other caller is
# refused even when it sends the identity header. Fill in once NCSA names the proxy.
PACELAB_TRUSTED_PROXIES=
# member (default): a NetID with no grant may sign in and sees no laboratory until one
# is assigned. none: such a NetID is turned away with 403.
PACELAB_UNLISTED_ROLE=
